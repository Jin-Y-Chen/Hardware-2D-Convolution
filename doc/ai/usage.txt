    Q: how to add package in verilog

    R: // rtl/conv_pkg.sv
    package conv_pkg;
        parameter int WIDTH = 16;
        parameter int ACCW  = 48;

        typedef logic signed [WIDTH-1:0] data_t;
        typedef logic signed [ACCW-1:0]  acc_t;
    endpackage

    import conv_pkg::*;          // whole package

    module mac (
        input  data_t input0,
        ...
    );
        acc_t acc;
    endmodule
